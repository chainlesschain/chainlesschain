import fs from 'node:fs';import path from 'node:path';import crypto from 'node:crypto';import {spawn,spawnSync} from 'node:child_process';import assert from 'node:assert/strict';
const root=path.resolve('.tmp/fw-cached-code-experiment');fs.mkdirSync(root,{recursive:true});

 run(process.execPath,[config.client,input,'before']);

 assert.equal(hash(),expected);
