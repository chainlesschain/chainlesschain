import { describe, it, expect } from "vitest";
import { inspectImageBudget } from "../src/chat/image-decode-budget.js";

// Fixture data (opaque bytes, not shared with any existing suite).
const gifB64 =
  "R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7";

// Builds a GIF canvas of `canvas` x `canvas` px carrying `count` delta frames,
// each a 1x1 image placed at (0,0). Only the logical-screen size is scaled, so
// every individual frame stays far below the 40 megapixel single-image cap.
function multiFrameGif(count, canvas) {
  const gif = Buffer.from(gifB64, "base64");
  const header = Buffer.from(gif.subarray(0, 19));
  header.writeUInt16LE(canvas, 6);
  header.writeUInt16LE(canvas, 8);
  return Buffer.concat([
    header,
    ...Array.from({ length: count }, () => gif.subarray(19, -1)),
    Buffer.from([0x3b]),
  ]);
}

describe("cumulative animation frame budget (admission before decode)", () => {
  it("rejects multi-frame accumulation that exceeds the 40M canvas budget", () => {
    // 1000x1000 canvas x 41 frames = 41,000,000 > 40,000,000 even though each
    // delta frame is only 1x1. Rejected synchronously, i.e. before any decode.
    expect(() => inspectImageBudget(multiFrameGif(41, 1000))).toThrow(
      "decoded canvas pixels",
    );
  });

  it("retains the exact legal cumulative boundary of 40M decoded canvas pixels", () => {
    // 1000x1000 x 40 frames = 40,000,000, the inclusive upper bound.
    expect(inspectImageBudget(multiFrameGif(40, 1000))).toMatchObject({
      format: "gif",
      frames: 40,
      decodedPixels: 40_000_000,
    });
  });

  it("accumulates tiny delta frames against the full canvas budget", () => {
    // A legal single canvas (4000x4000 = 16M <= 40M) becomes illegal once 3
    // frames are composited: 16M * 3 = 48M > 40M. Confirms the cumulative
    // accounting uses the canvas, not each small frame.
    expect(() => inspectImageBudget(multiFrameGif(3, 4000))).toThrow(
      "decoded canvas pixels",
    );
  });

  it("retains the 200-frame cap at the boundary and rejects one frame past it", () => {
    // 200 frames of 447x447 = 39,961,800 <= 40M: legal on both axes.
    expect(inspectImageBudget(multiFrameGif(200, 447))).toMatchObject({
      frames: 200,
      decodedPixels: 39_961_800,
    });
    // 201 frames of 1x1 = 201 pixels: only the frame-count cap (201 > 200)
    // triggers rejection, isolating the frame-count rule from the pixel cap.
    expect(() => inspectImageBudget(multiFrameGif(201, 1))).toThrow(
      "200 frames",
    );
  });

  it("retains a single legal large canvas and rejects one pixel past 40M", () => {
    // 6324x6324 = 39,992,976: largest single canvas under the 40M limit.
    expect(inspectImageBudget(multiFrameGif(1, 6324))).toMatchObject({
      frames: 1,
      decodedPixels: 39_992_976,
    });
    // 6325x6325 = 40,005,625: over the single-image limit.
    expect(() => inspectImageBudget(multiFrameGif(1, 6325))).toThrow(
      "megapixels",
    );
  });

  it("retains a plain non-animated image without fabricating extra canvases", () => {
    // 1x1 GIF with a single frame: still accepted with no extra fallback canvas.
    expect(inspectImageBudget(multiFrameGif(1, 1))).toMatchObject({
      format: "gif",
      frames: 1,
      extraCanvases: 0,
      decodedPixels: 1,
    });
  });
});
