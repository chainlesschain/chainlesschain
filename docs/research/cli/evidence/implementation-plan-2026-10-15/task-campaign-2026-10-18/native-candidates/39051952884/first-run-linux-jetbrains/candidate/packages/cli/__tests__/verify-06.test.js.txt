import { describe, it, expect, afterEach, vi } from "vitest";
import { Orchestrator, TASK_STATUS } from "../src/lib/orchestrator.js";

// Independent regression coverage for the paid-decomposition preflight in
// orchestrator.addTask (invoked from orchestrate.js): a rejected backend must
// terminate the task *before* decomposition, Run/evolution creation and any
// notification. No source text is asserted and no existing suite is imported.

describe("verify-06: paid-decomposition preflight", () => {
  afterEach(() => {
    vi.restoreAllMocks();
  });

  /** Orchestrator whose router exposes only opaque external CLIs (claude/codex). */
  function buildRejectedBackendOrchestrator({ notifyStart } = {}) {
    return new Orchestrator({
      agents: { backends: [{ type: "claude" }, { type: "codex" }] },
      notifier: {
        isConfigured: true,
        notifyStart: notifyStart || vi.fn(),
      },
      evolutionCompositionFactory: vi.fn(),
    });
  }

  it("rejects a CLI-only backend before decomposition, Run creation or notification", async () => {
    const notifyStart = vi.fn();
    const orch = buildRejectedBackendOrchestrator({ notifyStart });
    const decompose = vi.spyOn(orch, "_decompose");
    const compose = orch._evolutionCompositionFactory;

    await expect(orch.addTask("fix bug", { cwd: "/tmp" })).rejects.toMatchObject({
      code: "AGENT_ROUTER_EXTERNAL_MODEL_INGRESS_UNATTESTED",
    });

    // Preflight must terminate before any paid work happens.
    expect(decompose).not.toHaveBeenCalled();
    expect(compose).not.toHaveBeenCalled();
    expect(notifyStart).not.toHaveBeenCalled();
    expect(orch._tasks.size).toBe(0);
  });

  it("terminates before notification even when notify is requested", async () => {
    const notifyStart = vi.fn();
    const orch = buildRejectedBackendOrchestrator({ notifyStart });
    const decompose = vi.spyOn(orch, "_decompose");

    await expect(
      orch.addTask("sensitive task", { cwd: "/tmp", notify: true }),
    ).rejects.toMatchObject({
      code: "AGENT_ROUTER_EXTERNAL_MODEL_INGRESS_UNATTESTED",
    });

    expect(decompose).not.toHaveBeenCalled();
    expect(notifyStart).not.toHaveBeenCalled();
    expect(orch.status().tasks).toEqual([]);
  });

  it("rejects before the router ever dispatches subtasks", async () => {
    const orch = buildRejectedBackendOrchestrator();
    const dispatch = vi.spyOn(orch._router, "dispatch");
    const decompose = vi.spyOn(orch, "_decompose");

    await expect(orch.addTask("task")).rejects.toMatchObject({
      code: "AGENT_ROUTER_EXTERNAL_MODEL_INGRESS_UNATTESTED",
    });
    expect(decompose).not.toHaveBeenCalled();
    expect(dispatch).not.toHaveBeenCalled();
  });

  it("negative control: a non-CLI backend passes the preflight", () => {
    const orch = new Orchestrator({
      cwd: "/tmp",
      agents: { backends: [{ type: "claude", isCLI: false }] },
    });
    // A non-CLI backend satisfies the preflight gate: no throw.
    expect(orch._router.assertDispatchAvailable()).toBeUndefined();
  });

  it("negative control: no backends at all is also rejected by the preflight", async () => {
    const orch = new Orchestrator({
      agents: { backends: [] },
    });
    const decompose = vi.spyOn(orch, "_decompose");

    await expect(orch.addTask("task")).rejects.toMatchObject({
      code: "AGENT_ROUTER_NO_BACKENDS",
    });
    expect(decompose).not.toHaveBeenCalled();
  });
});
