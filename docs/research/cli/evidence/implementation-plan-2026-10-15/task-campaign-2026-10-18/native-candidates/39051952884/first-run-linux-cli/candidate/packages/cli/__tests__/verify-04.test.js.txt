import { describe, it, expect } from "vitest";
import { estimateCost, priceRollup } from "../src/lib/llm-pricing.js";

// Request-level cache and long-context pricing contract.
//
// GPT6_PRICING_TERMS (applied to gpt-6-* documented OpenAI models):
//   cacheReadMultiplier = 0.1, cacheWriteMultiplier = 1.25
//   longContext.threshold = 272000, inputMultiplier = 2, outputMultiplier = 1.5
//   serviceMultipliers: standard/default/auto = 1, batch/flex = 0.5, fast = 2
//
// estimateCost applies these tiers per request (promptTokens = requestInputTokens
// ?? inputTokens + cacheReadTokens + cacheCreationTokens), so the long-context
// threshold and the cache discounts must NEVER be applied to a summed aggregate.

const ASTRA = { provider: "openai", model: "gpt-6-astra", serviceTier: "standard" };

describe("request-level cache & long-context pricing (verify-04)", () => {
  it("applies the cache-read discount (0.1x) per request, not to an average", () => {
    const cached = estimateCost({ ...ASTRA, inputTokens: 100000, outputTokens: 10000, cacheReadTokens: 100000 });
    const uncached = estimateCost({ ...ASTRA, inputTokens: 100000, outputTokens: 10000, cacheReadTokens: 0 });
    // Uncached: input 100000@$10 + output 10000@$50 = 1.5
    expect(uncached.totalCost).toBeCloseTo(1.5, 10);
    // Cached: the cached 100000 input is billed at 10% => adds 0.1
    expect(cached.cacheReadCost).toBeCloseTo((100000 / 1e6) * 10 * 0.1, 10);
    expect(cached.cacheReadCost).toBeCloseTo(0.1, 10);
    expect(cached.totalCost).toBeCloseTo(1.6, 10);
    // Total is exactly the sum of per-request costs; no cross-request averaging.
    expect(cached.totalCost + uncached.totalCost).toBeCloseTo(3.1, 10);
  });

  it("never derives the long-context tier from a summed aggregate (negative control)", () => {
    // Two requests, each 150000 prompt tokens (below the 272000 threshold).
    // Their sum (300000) exceeds the threshold, but no single request does.
    const a = estimateCost({ ...ASTRA, inputTokens: 150000, outputTokens: 1000 });
    const b = estimateCost({ ...ASTRA, inputTokens: 150000, outputTokens: 1000 });
    // Each request: 150000@$10 + 1000@$50 = 1.55 (no 2x/1.5x long-context tier)
    expect(a.totalCost).toBeCloseTo(1.55, 10);
    expect(b.totalCost).toBeCloseTo(1.55, 10);
    expect(a.totalCost + b.totalCost).toBeCloseTo(3.1, 10);
    // A wrong aggregate sum (300000 > 272000 -> 2x input, 1.5x output) = 6.15.
    // The per-request contract must be strictly cheaper here.
    expect(a.totalCost + b.totalCost).toBeLessThan(6.15);
  });

  it("applies the long-context tier when a single request crosses the threshold", () => {
    const long = estimateCost({ ...ASTRA, inputTokens: 300000, outputTokens: 1000 });
    // 300000 > 272000 -> input x2, output x1.5.
    expect(long.inputCost).toBeCloseTo((300000 / 1e6) * 10 * 2, 10);
    expect(long.outputCost).toBeCloseTo((1000 / 1e6) * 50 * 1.5, 10);
    expect(long.totalCost).toBeCloseTo(6.075, 10);
  });

  it("bills cache writes at 1.25x the input rate, per request", () => {
    const write = estimateCost({ ...ASTRA, inputTokens: 0, outputTokens: 0, cacheCreationTokens: 100000 });
    // 100000 @ $10 * 1.25 (promptTokens = 100000, below long-context threshold).
    expect(write.cacheCreationCost).toBeCloseTo((100000 / 1e6) * 10 * 1.25, 10);
    expect(write.cacheCreationCost).toBeCloseTo(1.25, 10);
    expect(write.totalCost).toBeCloseTo(1.25, 10);
  });

  it("applies the service-tier multiplier per request (batch = half of standard)", () => {
    const base = { ...ASTRA, inputTokens: 100000, outputTokens: 10000, cacheReadTokens: 0 };
    const standard = estimateCost(base);
    const batch = estimateCost({ ...base, serviceTier: "batch" });
    const fast = estimateCost({ ...base, serviceTier: "fast" });
    expect(standard.totalCost).toBeCloseTo(1.5, 10);
    expect(batch.totalCost).toBeCloseTo(0.75, 10); // 0.5x
    expect(fast.totalCost).toBeCloseTo(3.0, 10); // 2x
    // Unknown tiers are not silently billed at standard — they are unpriced.
    const bogus = estimateCost({ ...base, serviceTier: "not-a-tier" });
    expect(bogus.matched).toBe(false);
    expect(bogus.totalCost).toBe(0);
  });

  it("priceRollup keeps per-request rows instead of re-deriving a long-context aggregate", () => {
    const rollup = priceRollup({
      byModel: [
        { provider: "openai", model: "gpt-6-astra", inputTokens: 150000, outputTokens: 1000, serviceTier: "standard" },
        { provider: "openai", model: "gpt-6-astra", inputTokens: 150000, outputTokens: 1000, serviceTier: "standard" },
      ],
    });
    // Both rows stay at the per-request price (no long-context on the 300000 sum).
    expect(rollup.byModel).toHaveLength(2);
    expect(rollup.byModel[0].cost.totalCost).toBeCloseTo(1.55, 10);
    expect(rollup.byModel[1].cost.totalCost).toBeCloseTo(1.55, 10);
    expect(rollup.byModel[0].cost.totalCost + rollup.byModel[1].cost.totalCost).toBeCloseTo(3.1, 10);
  });
});
