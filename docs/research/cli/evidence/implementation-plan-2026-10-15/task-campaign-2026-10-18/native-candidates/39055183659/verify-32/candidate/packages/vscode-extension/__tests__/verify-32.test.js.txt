import { describe, it, expect, vi } from "vitest";
import {
  createImagePreviewGate,
  drawImagePreview,
} from "../src/chat/image-preview-gate.js";

// Regression coverage for "preview cancellation ineffective": when a page is
// switched or a preview is cancelled, a late-arriving preview must not be
// written to the new page, while the current page still renders normally.
// These tests pin the correct behaviour of the gate and the preview drawer.

describe("gate: late preview is disposed, not delivered, after cancel", () => {
  it("closes late bitmaps and rejects the cancelled batch so nothing is drawn", async () => {
    const worker = { postMessage() {}, terminate() {} };
    const gate = createImagePreviewGate({ workerFactory: () => worker });
    const pending = gate.validate([{ data: "old" }]);
    const rejected = expect(pending).rejects.toThrow("cancelled");
    gate.cancel();
    const lateBitmap = { close: vi.fn() };
    // A result from the previous batch arrives after cancellation.
    worker.onmessage({
      data: { results: [{ thumbnails: [{ bitmap: lateBitmap }] }] },
    });
    await rejected;
    // The late preview is reclaimed, so it can never reach a canvas.
    expect(lateBitmap.close).toHaveBeenCalledOnce();
  });

  it("switching the visible batch disposes the old result and keeps the current one drawable", async () => {
    const workers = [];
    const gate = createImagePreviewGate({
      workerFactory: () => {
        const w = { postMessage: vi.fn(), terminate: vi.fn() };
        workers.push(w);
        return w;
      },
    });
    const first = gate.validate([{ data: "old" }]);
    const oldRejected = expect(first).rejects.toThrow("cancelled");
    const second = gate.validate([{ data: "new" }]);

    const oldBitmap = { close: vi.fn() };
    workers[0].onmessage({
      data: { results: [{ thumbnails: [{ bitmap: oldBitmap }] }] },
    });
    const newBitmap = { close: vi.fn() };
    workers[1].onmessage({
      data: { results: [{ thumbnails: [{ bitmap: newBitmap }] }] },
    });

    await oldRejected;
    const current = await second;
    // The late preview from the old page is closed and never written.
    expect(oldBitmap.close).toHaveBeenCalledOnce();
    // The current page's preview survives to be drawn.
    expect(current[0].thumbnails[0].bitmap).toBe(newBitmap);
    expect(newBitmap.close).not.toHaveBeenCalled();
  });

  it("closes late bitmaps delivered after the decode deadline", async () => {
    vi.useFakeTimers();
    try {
      const worker = { postMessage: vi.fn(), terminate: vi.fn() };
      const gate = createImagePreviewGate({
        timeoutMs: 25,
        workerFactory: () => worker,
      });
      const result = gate.validate([{ data: "stuck" }]);
      const rejected = expect(result).rejects.toThrow("budget");
      await vi.advanceTimersByTimeAsync(25);
      await rejected;
      const lateBitmap = { close: vi.fn() };
      worker.onmessage({
        data: { results: [{ thumbnails: [{ bitmap: lateBitmap }] }] },
      });
      // Arriving after the deadline, the preview is reclaimed, not rendered.
      expect(lateBitmap.close).toHaveBeenCalledOnce();
    } finally {
      vi.useRealTimers();
    }
  });
});

describe("drawImagePreview: cancelled preview stops writing, current page keeps rendering", () => {
  function drawingCanvas() {
    const context = { clearRect() {}, drawImage: vi.fn() };
    return { canvas: { getContext: () => context }, context };
  }

  it("stops writing further animation frames after the preview is stopped", async () => {
    vi.useFakeTimers();
    try {
      const { canvas, context } = drawingCanvas();
      const stop = drawImagePreview(canvas, {
        repetitions: Infinity,
        thumbnails: [
          { bitmap: "frame-a", duration: 1 },
          { bitmap: "frame-b", duration: 1 },
        ],
      });
      expect(context.drawImage).toHaveBeenLastCalledWith("frame-a", 0, 0);
      await vi.advanceTimersByTimeAsync(1);
      expect(context.drawImage).toHaveBeenLastCalledWith("frame-b", 0, 0);
      stop();
      await vi.advanceTimersByTimeAsync(200);
      // A cancelled preview writes no further frames to the (new) page.
      expect(context.drawImage).toHaveBeenCalledTimes(2);
    } finally {
      vi.useRealTimers();
    }
  });

  it("keeps animating the current, non-stopped preview until its repetition budget", async () => {
    vi.useFakeTimers();
    try {
      const { canvas, context } = drawingCanvas();
      drawImagePreview(canvas, {
        repetitions: 0,
        thumbnails: [
          { bitmap: "frame-a", duration: 1 },
          { bitmap: "frame-b", duration: 1 },
        ],
      });
      await vi.advanceTimersByTimeAsync(1);
      expect(context.drawImage).toHaveBeenCalledTimes(2);
      // With repetitions 0 the loop completes after one full cycle.
      await vi.advanceTimersByTimeAsync(100);
      expect(context.drawImage).toHaveBeenCalledTimes(2);
    } finally {
      vi.useRealTimers();
    }
  });
});

describe("gate: a stale worker failure never rejects the current batch", () => {
  it("ignores an old worker's error and delivers only the current page result", async () => {
    const workers = [];
    const gate = createImagePreviewGate({
      workerFactory: () => {
        const w = { postMessage: vi.fn(), terminate: vi.fn() };
        workers.push(w);
        return w;
      },
    });
    const first = gate.validate([{ data: "old" }]);
    const oldRejected = expect(first).rejects.toThrow("cancelled");
    const second = gate.validate([{ data: "new" }]);

    workers[0].onerror();
    const newBitmap = { close: vi.fn() };
    workers[1].onmessage({
      data: { results: [{ thumbnails: [{ bitmap: newBitmap }] }] },
    });

    await oldRejected;
    const current = await second;
    expect(current[0].thumbnails[0].bitmap).toBe(newBitmap);
    expect(newBitmap.close).not.toHaveBeenCalled();
  });
});
