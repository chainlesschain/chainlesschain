import { describe, expect, it } from "vitest";
import {
  createEvalComparison,
  createEvalHistoryRecord,
  EVAL_CONTEXT_SCHEMA,
  EVAL_EXECUTION_PROTOCOL,
  evaluateStrictEvalGate,
} from "../src/lib/eval/evidence.js";

const NOW = Date.parse("2026-09-12T12:00:00.000Z");
const D = (character) => `sha256:${character.repeat(64)}`;

function context(overrides = {}) {
  return {
    schema: EVAL_CONTEXT_SCHEMA,
    provider: "openai",
    model: "test-model",
    modelRevision: "test-model@revision-1",
    environmentDigest: D("a"),
    permissionDigest: D("b"),
    inferenceDigest: D("c"),
    ...overrides,
  };
}

function comparison(overrides = {}) {
  return createEvalComparison({
    suite: "builtin",
    corpusDigest: D("d"),
    provider: overrides.provider ?? "openai",
    model: overrides.model ?? "test-model",
    context: context(overrides.context ?? {}),
    platform: overrides.platform ?? "linux",
    arch: overrides.arch ?? "x64",
    node: overrides.node ?? "v22.22.2",
  });
}

function result(id, pass) {
  return {
    id,
    pass,
    artifactCheckPassed: pass,
    executionSucceeded: true,
    agentOk: true,
    executionEvidence: {
      protocol: EVAL_EXECUTION_PROTOCOL,
      exitCode: 0,
      signal: null,
      terminalVerified: true,
      observedFallback: false,
    },
    error: null,
    detail: "checked",
    changedFiles: [],
    unrelatedChanges: [],
  };
}

function record(index, cmp, pairs = [
  ["a", true],
  ["b", true],
]) {
  const passed = pairs.filter(([, pass]) => pass).length;
  return createEvalHistoryRecord(
    {
      total: pairs.length,
      passed,
      failed: pairs.length - passed,
      passRate: passed / pairs.length,
      unrelatedChangeRate: 0,
      results: pairs.map(([id, pass]) => result(id, pass)),
    },
    {
      comparison: cmp,
      runId: `run-${index}`,
      ranAt: new Date(NOW - (3 - index) * 1000).toISOString(),
    },
  );
}

const gate = (runs) =>
  evaluateStrictEvalGate({ runs, issues: [] }, { now: NOW });

describe("verify-24: strict comparison identity isolation", () => {
  it("accepts an improvement only when model/permission/runtime identity is unchanged", () => {
    const baseline = record(1, comparison());
    const candidate = record(2, comparison());
    expect(gate([baseline, candidate])).toMatchObject({
      passed: true,
      status: "PASS",
      baselineRunId: "run-1",
      candidateRunId: "run-2",
    });
  });

  it.each([
    ["model", comparison({ model: "other-model", context: { model: "other-model" } })],
    ["provider", comparison({ provider: "anthropic", context: { provider: "anthropic" } })],
    ["permission digest", comparison({ context: { permissionDigest: D("e") } })],
    ["runtime", comparison({ node: "v24.0.0" })],
  ])(
    "does not count a same-task run as comparable when %s identity differs",
    (_label, otherCmp) => {
      // Identical task population, results and verdicts; only the declared
      // model/permission/runtime identity differs from the baseline.
      const baseline = record(1, comparison());
      const candidate = record(2, otherCmp);
      const out = gate([baseline, candidate]);
      expect(out.passed).toBe(false);
      expect(out.reasons).toContain("comparison_identity_changed");
    },
  );

  it("still requires an identical task population even when identity matches", () => {
    const baseline = record(1, comparison());
    const candidate = record(2, comparison(), [
      ["a", true],
      ["b", true],
      ["c", true],
    ]);
    const out = gate([baseline, candidate]);
    expect(out.passed).toBe(false);
    expect(out.reasons).toContain("task_population_changed");
  });
});
