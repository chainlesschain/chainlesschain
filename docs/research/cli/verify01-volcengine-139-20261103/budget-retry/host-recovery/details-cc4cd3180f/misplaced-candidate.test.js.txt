import { describe, it, expect, beforeEach, afterEach, vi } from "vitest";
import { Orchestrator } from "../../src/lib/orchestrator.js";
import { _deps as bridgeDeps } from "../../src/lib/claude-code-bridge.js";

/**
 * verify-06 — 付费分解前预检 (pre-paid-decomposition preflight)
 *
 * Regression coverage for the CLI-only agent-backend rejection guard.
 *
 * When the orchestrator is configured exclusively with CLI-only agent
 * backends (e.g. claude / codex), adding a task must be rejected with
 * AGENT_ROUTER_EXTERNAL_MODEL_INGRESS_UNATTESTED *before* any paid
 * decomposition (this._decompose) or any notification is emitted.
 *
 * A positive control uses an API/remote backend (non-CLI, isCLI:false)
 * to prove the same guard permits the task to reach decomposition.
 */

function makeChildProcess(stdout = "", exitCode = 0) {
  const { EventEmitter } = require("events");
  const proc = new EventEmitter();
  proc.stdout = new EventEmitter();
  proc.stderr = new EventEmitter();
  proc.kill = vi.fn();
  process.nextTick(() => {
    if (stdout) proc.stdout.emit("data", Buffer.from(stdout));
    proc.emit("close", exitCode);
  });
  return proc;
}

function buildOrchestrator(backends) {
  return new Orchestrator({
    agents: { backends },
    maxParallel: 2,
    maxRetries: 0,
    ciCommand: "exit 0",
  });
}

// A deterministic decomposition payload so a non-CLI backend can proceed.
const DECOMPOSED = JSON.stringify({
  type: "decomposition",
  subtasks: [
    { id: "sub-1", description: "fix the thing", execute: { run: "true" } },
  ],
});

describe("paid decomposition preflight for CLI-only backends", () => {
  beforeEach(() => {
    bridgeDeps.spawn = vi.fn(() => makeChildProcess());
    vi.spyOn(bridgeDeps, "spawn");
  });
  afterEach(() => {
    vi.restoreAllMocks();
  });

  it("rejects a CLI-only backend before decomposition or notification", async () => {
    const orch = buildOrchestrator([
      { type: "claude" },
      { type: "codex" },
    ]);

    const decomposeSpy = vi.spyOn(orch, "_decompose");
    let notifySpy = vi.fn();
    if (orch.notifier) {
      notifySpy = vi.spyOn(orch.notifier, "notifyStart");
    } else {
      orch.notifier = { isConfigured: true, notifyStart: notifySpy };
    }

    await expect(orch.addTask("fix the thing")).rejects.toMatchObject({
      code: "AGENT_ROUTER_EXTERNAL_MODEL_INGRESS_UNATTESTED",
    });

    // The guard fires before decomposition and before the start notification.
    expect(decomposeSpy).not.toHaveBeenCalled();
    expect(notifySpy).not.toHaveBeenCalled();

    // No task was registered either.
    expect(orch._tasks.size).toBe(0);
  });

  it("permits a non-CLI API backend to reach decomposition (positive control)", async () => {
    const orch = buildOrchestrator([{ type: "anthropic" }]);

    // Stub the LLM ingress so _decompose can resolve deterministically.
    orch._chat = vi.fn(async () => DECOMPOSED);

    const decomposeSpy = vi.spyOn(orch, "_decompose").mockImplementation(
      async (task) => {
        task.status = "decomposed";
        return task;
      },
    );

    const task = await orch.addTask("fix the thing");

    expect(task).toBeDefined();
    expect(decomposeSpy).toHaveBeenCalledTimes(1);
    expect(orch._tasks.size).toBe(1);
  });
});
